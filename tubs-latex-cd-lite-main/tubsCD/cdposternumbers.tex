% define line distances (zeilenabstände)
\pgfmathsetmacro{\ZAheadlinebiggerfontsize}{\headlinebiggerfontsize * 1.10}%
\pgfmathsetmacro{\ZAheadlinesmallerfontsize}{\headlinesmallerfontsize * 1.10}%
\pgfmathsetmacro{\ZAsubheadlinefontsize}{\subheadlinefontsize * 1.20}%
\pgfmathsetmacro{\ZAinstitutefontsize}{\institutefontsize * 1.20}%
\pgfmathsetmacro{\ZAposternormaltextfontsize}{\posternormaltextfontsize * 1.30}%
\pgfmathsetmacro{\ZApostercopytextfontsize}{\postercopytextfontsize * 1.40}%

% define fonts
\newcommand{\tubsFontHeadline}{%
    \vspace*{\borderwidth mm}% headlines get extra leading space due to toolbox definition
    \fontsize{\headlinebiggerfontsize}{\ZAheadlinebiggerfontsize}%
    \selectfont%
}%
\newcommand{\tubsFontHeadlineSmall}{%
    \vspace*{\borderwidth mm}% headlines get extra leading space due to toolbox definition
    \fontsize{\headlinesmallerfontsize}{\ZAheadlinesmallerfontsize}%

}%
